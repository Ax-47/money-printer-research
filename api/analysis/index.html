    analysis¶

    money_printer_research.analysis.feature_importance ¶

    Feature selection for the H/D/A model on model_frame.csv.

    Features come from an explicit allowlist of groups (FEATURE_GROUPS), each built from a pre-match source; nothing outside it can reach the model, so the result of the match (goals, xG, is_result) cannot leak in. A group is one feature for both sides plus their difference, so a side is never kept without the other.

    Selection is greedy and forward, by group, starting from Elo: add the group that lowers the walk-forward log loss most, until no group lowers it by MIN_GAIN. Walk-forward means each season in VALID_SEASONS is scored by a model fitted on the seasons before it only. The seasons from TEST_FROM on are touched once, at the end, to compare the selection with Elo alone and with every group.

    The model is a multinomial logistic regression on standardised features, nulls filled with the training median: on this data it beats gradient-boosted trees, which overfit the few hundred games a feature's signal rests on.

    CLASSES = ['A', 'D', 'H'] module-attribute ¶

    Sorted, the order of predict_proba's columns.

    FEATURE_GROUPS = {'elo': ['elo_home', 'elo_away', 'elo_diff'], 'league': [_league_column(lg) for lg in LEAGUES], 'fatigue': _sides('fatigue'), 'min_7d': _sides('min_7d'), **{s: _sides(s, diff=s not in ('prev_europe_away', 'next_is_europe')) for s in SCHEDULE}, 'formation': [f'{p}_{x}' for p in ('home', 'away') for x in ('defenders', 'midfielders', 'forwards', 'lines')], 'n_low_min': _sides('n_low_min', diff=False), **{f'player_{f}': _sides(f'{f}_mean') for f in PLAYER_SLOT_FEATURES}} module-attribute ¶

    Every candidate feature, by group. Player slots enter as each side's starters' mean.

    MIN_GAIN = 0.0005 module-attribute ¶

    Smallest drop in mean walk-forward log loss for which a group is added.

    POST_MATCH = re.compile('^(full_time_.*|half_time_.*|(home|away)_(goals|xg)|tm_(home|away)_goals|is_result)$') module-attribute ¶

    Known only after the match. No feature group may contain one (checked in load).

    SELECTED = ['elo', 'player_form90', 'games_14d_all', 'player_prev_min'] module-attribute ¶

    def fit_score(train: pl.DataFrame, test: pl.DataFrame, groups: list[str]) -> float:
    """Log loss on `test` of make_model fitted on `train` with the columns of `groups`."""
    cols = columns(groups)
    model = make_model().fit(train.select(cols).to_numpy(), train[TARGET].to_numpy())

    def forward_selection(
    df: pl.DataFrame,
    start: list[str] | None = None,
    min_gain: float = MIN_GAIN,
    seasons: tuple[int, ...] = VALID_SEASONS,
) -> pl.DataFrame:
    """Greedy forward selection by group; one row per step: the group added and the loss."""
    chosen = ["elo"] if start is None else list(start)
    best = walk_forward(df, chosen, seasons)
    steps: list[dict[str, Any]] = [{"added": "+".join(chosen), "log_loss": best, "gain": None}]
    while rest := [g for g in FEATURE_GROUPS if g not in chosen]:
        scores = {g: walk_forward(df, [*chosen, g], seasons) for g in rest}
        group = min(scores, key=scores.__getitem__)
        if best - scores[group] < min_gain:
            break
        chosen.append(group)
        steps.append({"added": group, "log_loss": scores[group], "gain": best - scores[group]})
        best = scores[group]
    return pl.DataFrame(
        steps, schema={"added": pl.String, "log_loss": pl.Float64, "gain": pl.Float64}
    )

    def held_out(df: pl.DataFrame, candidates: dict[str, list[str]]) -> pl.DataFrame:
    """Log loss on the seasons from TEST_FROM of each candidate, fitted on the seasons before."""
    season = pl.col("season_start")
    train, test = df.filter(season < TEST_FROM), df.filter(season >= TEST_FROM)
    rows = [{"features": "class prior", "log_loss": prior_score(train, test)}]

    def load(path: Path) -> pl.DataFrame:

    def main(path: Path) -> None:
    """Print the forward selection, then the held-out comparison."""
    df = load(path)
    print(f"{df.height} games, {len(FEATURE_GROUPS)} feature groups, test from {TEST_FROM}\n")
    steps = forward_selection(df.filter(pl.col("season_start") < TEST_FROM))
    chosen = [g for s in steps["added"] for g in s.split("+")]
    with pl.Config(float_precision=4):
        print("Forward selection, walk-forward log loss over", VALID_SEASONS)
        print(steps)
        print("\nHeld out:")
        print(
            held_out(df, {"elo": ["elo"], "selected": chosen, "every group": list(FEATURE_GROUPS)})
        )
    if chosen != SELECTED:
        print(f"\nselection differs from SELECTED: {chosen}")

    def prior_score(train: pl.DataFrame, test: pl.DataFrame) -> float:
    """Log loss of predicting every game with the training shares of A, D and H."""
    shares = [float((train[TARGET] == c).mean()) for c in CLASSES]  # pyright: ignore[reportArgumentType]
    return float(log_loss(test[TARGET].to_numpy(), [shares] * test.height, labels=CLASSES))

    def walk_forward(
    df: pl.DataFrame, groups: list[str], seasons: tuple[int, ...] = VALID_SEASONS
) -> float:
    """Mean log loss over `seasons`, each scored by a model fitted on the seasons before it."""
    season = pl.col("season_start")
    return float(
        np.mean(
            [fit_score(df.filter(season < s), df.filter(season == s), groups) for s in seasons]
        )
    )

    def with_derived(df: pl.DataFrame) -> pl.DataFrame:
    """Add the derived columns FEATURE_GROUPS names: slot means, diffs, league flags.

    <f>_mean is the mean of a side's starter slots (nulls skipped), n_low_min the number
    of its starters under MIN_PREV_MIN previous minutes, <name>_diff home minus away.
    A diff whose h_/a_ columns are not in `df` is left out (predict's fixtures carry only
    the features the saved model uses); selecting it later fails loudly.
    """
    slots = {
        (s, f): [c for c in df.columns if re.fullmatch(rf"{s}_player\d+_{f}", c)]
        for s in ("h", "a")
        for f in PLAYER_SLOT_FEATURES
    }
    df = df.with_columns(
        *((pl.col("league") == lg).cast(pl.Int8).alias(_league_column(lg)) for lg in LEAGUES),
        *(pl.mean_horizontal(cols).alias(f"{s}_{f}_mean") for (s, f), cols in slots.items()),
        *(
            pl.sum_horizontal(pl.col(c).is_null() for c in slots[s, "xg90"]).alias(
                f"{s}_n_low_min"
            )
            for s in ("h", "a")
        ),
    )
    diffs = {
        c.removesuffix("_diff")
        for cols in FEATURE_GROUPS.values()
        for c in cols
        if c.endswith("_diff")
    }
    present = [n for n in sorted(diffs - {"elo"}) if {f"h_{n}", f"a_{n}"} <= set(df.columns)]
    return df.with_columns(
        (pl.col(f"h_{n}") - pl.col(f"a_{n}")).alias(f"{n}_diff") for n in present
    )

    money_printer_research.analysis.modeling ¶

    H/D/A models on the selected features (feature_importance.SELECTED), compared and checked.

    Every model is fitted and scored the way feature selection was: walk-forward over VALID_SEASONS (each season predicted by a model fitted on the seasons before it), and once on the held-out seasons from TEST_FROM, fitted on everything before.

    The baseline is feature_importance.make_model, a multinomial logistic regression. It is compared with the same model recalibrated (sigmoid, 5-fold on the training seasons), gradient-boosted trees, and a goals model: one Poisson regression per side for the number of goals, turned into H/D/A by summing the independent score grid.

    Scores are log loss (what selection minimised), the ranked probability score (the usual one for football: it treats H, D, A as ordered, so calling a home win a draw costs less than calling it an away win), the multiclass Brier score and accuracy. calibration bins each class's predicted probability by decile and compares it with how often it happened.

    After the comparison, fit_final refits the logistic baseline on every season (the held-out ones too: they have been scored, and the newest games matter most) and save_model writes it with joblib to settings.modeling.out_file. load_model reads it back; SavedModel.predict takes model_frame rows and returns P(A), P(D), P(H) per game. uv run modeling then writes the learning curve (analysis.learning_curve).

    def predict(self, frame: pl.DataFrame) -> pl.DataFrame:
    """P(A), P(D), P(H) for each row of `frame` (model_frame columns), plus its game_id."""
    x = with_derived(frame).select(pl.col(self.columns).cast(pl.Float64)).to_numpy()

    def boosted_trees(train: pl.DataFrame, test: pl.DataFrame, cols: list[str]) -> np.ndarray:
    """Shallow, slow gradient boosting; trees take the nulls as they are.

    On more than 10,000 games scikit-learn stops early on a random 10% of the training
    games; random_state fixes that split, so the scores are the same on every run.
    """
    model = HistGradientBoostingClassifier(
        max_depth=3, learning_rate=0.03, max_iter=300, l2_regularization=1.0, random_state=0
    )
    return _fit_predict(model, train, test, cols)

    def calibration(y: np.ndarray, proba: np.ndarray, bins: int = 10) -> pl.DataFrame:
    """Per class and probability decile: games, mean predicted probability, observed rate."""
    frames = []
    for i, c in enumerate(CLASSES):
        p = proba[:, i]
        edges = np.quantile(p, np.linspace(0, 1, bins + 1))
        frames.append(
            pl.DataFrame(
                {
                    "class": c,
                    "bin": np.clip(np.searchsorted(edges, p, side="right") - 1, 0, bins - 1),
                    "predicted": p,
                    "observed": (y == c).astype(float),
                }
            )
        )
    return (
        pl.concat(frames)
        .group_by("class", "bin", maintain_order=True)
        .agg(pl.len().alias("games"), pl.col("predicted", "observed").mean())
        .sort("class", "bin")
    )

    def calibration_error(table: pl.DataFrame) -> pl.DataFrame:
    """Per class: expected calibration error, the games-weighted mean |predicted - observed|."""
    return table.group_by("class", maintain_order=True).agg(
        (
            ((pl.col("predicted") - pl.col("observed")).abs() * pl.col("games")).sum()
            / pl.col("games").sum()
        ).alias("ece")
    )

    def rps(y: np.ndarray, proba: np.ndarray) -> float:
    """Mean ranked probability score, outcomes ordered H, D, A (lower is better)."""
    order = [CLASSES.index(c) for c in ("H", "D", "A")]
    gap = np.cumsum(proba[:, order], 1) - np.cumsum(_onehot(y)[:, order], 1)
    return float(np.mean(np.sum(gap[:, :-1] ** 2, 1) / (len(CLASSES) - 1)))

    def score_grid_proba(home_rate: np.ndarray, away_rate: np.ndarray) -> np.ndarray:
    """P(A), P(D), P(H) for independent Poisson goals with these means, per game."""
    goals = np.arange(MAX_GOALS + 1)
    home = np.asarray(poisson.pmf(goals, home_rate[:, None]))
    away = np.asarray(poisson.pmf(goals, away_rate[:, None]))
    grid = home[:, :, None] * away[:, None, :]  # [game, home goals, away goals]
    proba = np.stack(
        [
            np.triu(grid, 1).sum((1, 2)),
            np.trace(grid, axis1=1, axis2=2),
            np.tril(grid, -1).sum((1, 2)),
        ],
        axis=1,
    )
    return proba / proba.sum(1, keepdims=True)

    Backtest: would betting on the model's +EV picks against the bookmakers have paid?

    Each season from settings.modeling.backtest_from is predicted walk-forward by the logistic baseline (fitted on the seasons before it, as in modeling), then priced with the 1X2 odds football-data.co.uk publishes per league and season, which collect_data downloads to settings.modeling.odds_dir (data_collection.scrape_odds).

    The odds are Pinnacle's closing prices: the sharpest widely quoted market, and what a bet placed just before kick-off would get. Where they are missing the next available is used: Pinnacle's early price, the market average at closing, then Bet365's early price (ODDS_PREFERENCE; analysis.odds explains early and closing).

    football-data names teams its own way ("Man United", "Paris SG"). team_names maps them to the match_features names by which team each one plays alongside on the same days, so no alias table is kept by hand.

    For every game the pick is the outcome with the highest expected value of a 1-unit stake, EV = p * odds - 1. The backtest table keeps every game with its pick, EV and profit, so a strategy "bet the pick when its EV is above t" is a filter on it (summary).

    A threshold picked by looking at the results flatters them, so two checks keep it honest: walk_forward_thresholds picks each season's threshold from the seasons before it only, and best_by_chance asks how often a model with no edge at all would show a best (league, threshold) cell as good as the best one here.

    def best_by_chance(
    table: pl.DataFrame,
    thresholds: Sequence[float] = SLIDER_THRESHOLDS,
    sims: int = 200,
    seed: int = 0,
) -> dict[str, float]:
    """How often the best (league, threshold) cell looks this good by luck alone.

    observed is the z-score of the best cell, where a cell is one league, or all of them,
    at one threshold. Each simulation replays the same picks at the same odds in a world
    where the model has no edge: every result is drawn from the bookmaker's margin-free
    probabilities. p_value is the share of simulations whose best cell scores at least
    as high as observed; far from 0, the best cell is no better than searching this many
    cells finds by chance.
    """
    leagues = table["league"].to_numpy()
    groups = [np.ones(table.height, bool), *(leagues == lg for lg in np.unique(leagues))]
    ev = table["ev"].to_numpy()
    pick_odds = _odds_of_pick(table)

    def best_z_score(won: np.ndarray) -> float:
        return _best_z_score(np.where(won, pick_odds - 1, -1.0), ev, groups, thresholds)

    observed = best_z_score((table["pick"] == table[TARGET]).to_numpy())

    implied = 1 / table.select(f"odds_{k}" for k in OUTCOMES).to_numpy()
    market_cdf = (implied / implied.sum(axis=1, keepdims=True)).cumsum(axis=1)
    pick = table["pick"].replace_strict({k: i for i, k in enumerate(OUTCOMES)}).to_numpy()
    rng = np.random.default_rng(seed)

    def no_edge_best_z_score() -> float:
        # A uniform draw lands past as many cumulative probabilities as the drawn
        # outcome's index: 0 is H, 1 is D, 2 is A.
        result = (rng.random(table.height)[:, None] > market_cdf).sum(axis=1)
        return best_z_score(result == pick)

    by_chance = np.array([no_edge_best_z_score() for _ in range(sims)])
    return {"observed": observed, "p_value": float(np.mean(by_chance >= observed))}

    pick is the outcome with the highest EV; profit is what 1 unit on it returned.
    """
    p = games.select(f"p_{k}" for k in OUTCOMES).to_numpy()
    odds = games.select(f"odds_{k}" for k in OUTCOMES).to_numpy()
    ev = p * odds - 1
    best = ev.argmax(axis=1)
    rows = np.arange(len(best))
    pick = np.array(OUTCOMES)[best]
    won = pick == games[TARGET].to_numpy()
    return games.with_columns(
        pick=pl.Series(pick),
        ev=pl.Series(ev[rows, best]),
        profit=pl.Series(np.where(won, odds[rows, best] - 1, -1.0)),
    )

    def summary(table: pl.DataFrame, min_ev: float, by: list[str] | None = None) -> pl.DataFrame:
    """Bets, hit rate, profit and ROI of betting 1 unit on every pick with ev > min_ev."""
    bets = table.filter(pl.col("ev") > min_ev)
    aggs = [
        pl.len().alias("bets"),
        (pl.col("profit") > 0).mean().alias("hit_rate"),
        pl.col("profit").sum().alias("profit"),
        pl.col("profit").mean().alias("roi"),
        (pl.col("profit").std() / pl.len().sqrt()).alias("roi_se"),
    ]
    return bets.group_by(by, maintain_order=True).agg(aggs).sort(by) if by else bets.select(aggs)

    def team_names(odds: pl.DataFrame, games: pl.DataFrame) -> pl.DataFrame:
    """league, fd_name, team: each football-data name and the games name it stands for.

    A name maps to the team it shares most game days with on the same side (home or
    away), in the same league and season. The other games of a match day add noise, but
    the right team shares every one of its days.
    """
    games = games.with_columns(day=pl.col("date").dt.date())
    pairs = pl.concat(
        odds.join(games, on=["league", "season_start", "day"]).select(
            "league", pl.col(f"fd_{side}").alias("fd_name"), pl.col(f"{side}_team").alias("team")
        )
        for side in ("home", "away")
    )
    return (
        pairs.group_by("league", "fd_name", "team")
        .len()
        .sort("len", "team", descending=[True, False])
        .group_by("league", "fd_name", maintain_order=True)
        .first()
        .drop("len")
    )

    def walk_forward(df: pl.DataFrame, seasons: list[int]) -> pl.DataFrame:
    """game_id with p_H, p_D, p_A for `seasons`, each from the logistic fitted on earlier ones."""
    cols, season = columns(SELECTED), pl.col("season_start")
    frames = []
    for s in seasons:
        test = df.filter(season == s)
        proba = logistic(df.filter(season < s), test, cols)
        frames.append(
            test.select("game_id").hstack(
                pl.DataFrame({f"p_{c}": proba[:, i] for i, c in enumerate(CLASSES)})
            )
        )
    return pl.concat(frames)

    def walk_forward_thresholds(
    table: pl.DataFrame, thresholds: Sequence[float] = SLIDER_THRESHOLDS
) -> pl.DataFrame:
    """Each season bet at a hindsight-free threshold (ThresholdSeasonCols).

    Before each season, pick the threshold that made the most on the seasons before it,
    as someone betting live would have to. The first season has nothing before it to
    choose from, so it is left out.
    """
    seasons = table["season_start"].to_numpy()
    ev = table["ev"].to_numpy()
    profit = table["profit"].to_numpy()

    rows = []
    for season in sorted(set(seasons.tolist()))[1:]:
        before = seasons < season
        chosen = max(thresholds, key=lambda t: profit[before & (ev > t)].sum())
        bets = profit[(seasons == season) & (ev > chosen)]
        roi = bets.mean() if bets.size else None
        rows.append((season, chosen, bets.size, bets.sum(), roi))
    table = pl.DataFrame(
        rows,
        schema={
            "season_start": pl.Int32,
            "min_ev": pl.Float64,
            "bets": pl.Int64,
            "profit": pl.Float64,
            "roi": pl.Float64,
        },
        orient="row",
    )
    return ThresholdSeasonCols.validate(table, lazy=True)
