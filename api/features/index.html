    features¶

    money_printer_research.features.elo ¶

    Double Poisson match model driven by pre-match Elo, with optional Dixon-Coles.

    Home and away goals are modelled as two Poisson variables whose log-rates are linear in the pre-match Elo difference. Outcome probabilities (H/D/A) come from the joint score grid. Everything is walk-forward: each season is predicted by a model fitted only on earlier seasons, and Elo ratings are always pre-match.

    ELO_SCALE = 400.0 module-attribute ¶

    Elo points per factor of 10 in the odds; also scales elo_diff into the regressors.

    EPS = 1e-12 module-attribute ¶

    Floor on probabilities before taking a log.

    MAX_GOALS = 10 module-attribute ¶

    Largest score per side in the grid; observed scores above it are capped to it.

    PROMOTED_GAP = 80.0 module-attribute ¶

    Points below base a promoted team starts at when no team went down (the first season).

    RHO_GRID = np.linspace(-0.25, 0.1, 36) module-attribute ¶

    Dixon-Coles rho values _fit_rho tries.

    EloParams(k=15.0, home_adv=40.0, season_regress=0.0, base=1500.0) dataclass ¶

    Elo settings.

    Tuned on 2005/06-2014/15 by walk-forward log loss. The surface is flat: K 10-20, home_adv 20-80 and regress 0-0.2 all land within 0.001.

    base = 1500.0 class-attribute instance-attribute ¶

    Mean rating.

    home_adv = 40.0 class-attribute instance-attribute ¶

    Elo points added to the home side in the expectation.

    k = 15.0 class-attribute instance-attribute ¶

    Update size.

    season_regress = 0.0 class-attribute instance-attribute ¶

    Share pulled back toward base at each season start.

    def evaluate(pred: DataFrame[PredictionCols]) -> dict[str, float]:
    """Log loss, ranked probability score and accuracy of p_home/p_draw/p_away."""
    p = pred.select(PredictionCols.p_home, PredictionCols.p_draw, PredictionCols.p_away).to_numpy()
    y = np.array([OUTCOMES.index(r) for r in pred[RESULT].to_list()])
    onehot = np.eye(len(OUTCOMES))[y]
    log_loss = float(-np.log(np.clip(p[np.arange(len(y)), y], EPS, None)).mean())
    # RPS: squared gaps of the cumulative distributions; the last one is always 0.
    rps = float((((np.cumsum(p, 1) - np.cumsum(onehot, 1))[:, :-1]) ** 2).sum(1).mean() / 2)
    acc = float((p.argmax(1) == y).mean())
    return {"log_loss": log_loss, "rps": rps, "accuracy": acc, "n": len(y)}

    def fixture_elo(
    played: DataFrame[EloInputCols], fixtures: pl.DataFrame, params: EloParams = DEFAULT_ELO
) -> pl.DataFrame:
    """Pre-match elo_home, elo_away and elo_diff for unplayed fixtures.

    `fixtures` has EloInputCols' season, match_date, home_team and away_team. The ratings
    are those after every played match; a fixture has no result, so it moves none. A
    fixture in a season the played matches have not reached starts that season first
    (promoted teams, regression toward base), as get_elo would.
    """
    played = DataFrame[EloInputCols](played.sort(EloCols.match_date))
    *_, rating, last_season_teams, current_season = _replay(played, params)
    cols = [EloCols.season, EloCols.home_team, EloCols.away_team]
    teams_by_season = _teams_by_season(pl.concat([played.select(cols), fixtures.select(cols)]))  # pyright: ignore[reportArgumentType]
    fixtures = fixtures.sort(EloCols.match_date)
    elo_home: list[float] = []
    elo_away: list[float] = []
    for season, home, away in fixtures.select(cols).iter_rows():
        if season != current_season:
            this_season_teams = teams_by_season[season]
            _start_season(rating, last_season_teams, this_season_teams, params)
            last_season_teams, current_season = this_season_teams, season
        elo_home.append(rating[home])
        elo_away.append(rating[away])
    return _with_diff(fixtures, elo_home, elo_away)

    def outcome_probs(grid: FloatArray) -> FloatArray:
    """(n, 3) probabilities in OUTCOMES order: home win, draw, away win."""
    shape = grid.shape[1:]
    home = np.tril(np.ones(shape), k=-1)  # h > a
    draw = np.eye(shape[0])
    away = np.triu(np.ones(shape), k=1)  # a > h
    return np.stack([(grid * m).sum(axis=(1, 2)) for m in (home, draw, away)], axis=1)

    def score_grid(
    lam: FloatArray, mu: FloatArray, rho: float = 0.0, max_goals: int = MAX_GOALS
) -> FloatArray:
    """Joint probability of each scoreline; grid[i, h, a] = P(home=h, away=a)."""
    goals = np.arange(max_goals + 1)
    # scipy's stubs type pmf as float; with array inputs it returns an (n, g) array.
    ph = _as_float(poisson.pmf(goals[None, :], lam[:, None]))
    pa = _as_float(poisson.pmf(goals[None, :], mu[:, None]))
    grid = ph[:, :, None] * pa[:, None, :]
    if rho:
        grid = grid * _dc_tau(lam, mu, rho, max_goals)
    return grid / grid.sum(axis=(1, 2), keepdims=True)

    @returns(EloInputCols)
def to_elo_input(matches: DataFrame[CleanedMatchCols]) -> pl.DataFrame:
    """cleaned_matches -> EloInputCols: season label, date only, full-time goals and result."""
    ei = EloInputCols
    start = pl.col(CleanedMatchCols.season_start)
    hg, ag = pl.col(CleanedMatchCols.home_goals), pl.col(CleanedMatchCols.away_goals)
    return matches.select(
        season_label(start).alias(ei.season),
        pl.col(CleanedMatchCols.date).dt.date().alias(ei.match_date),
        pl.col(CleanedMatchCols.home_team).alias(ei.home_team),
        pl.col(CleanedMatchCols.away_team).alias(ei.away_team),
        hg.alias(ei.full_time_home_goals),
        ag.alias(ei.full_time_away_goals),
        pl.when(hg > ag)
        .then(pl.lit("H"))
        .when(hg == ag)
        .then(pl.lit("D"))
        .otherwise(pl.lit("A"))
        .alias(ei.full_time_result),
    )

    def walk_forward(
    df: DataFrame[EloCols], test_seasons: list[str], dixon_coles: bool = True
) -> DataFrame[PredictionCols]:
    """Predict each test season with a model fitted on all earlier seasons."""
    seasons = sorted(df[EloCols.season].unique().to_list())
    predictions: list[pl.DataFrame] = []
    for season in test_seasons:
        earlier = [s for s in seasons if s < season]
        train = df.filter(pl.col(EloCols.season).is_in(earlier))
        test = df.filter(pl.col(EloCols.season) == season)
        predictions.append(_predict_season(train, test, dixon_coles))
    return PredictionCols.validate(pl.concat(predictions), lazy=True)

    def player_fatigue(player_matches: DataFrame[FatigueInputCols]) -> DataFrame[PlayerFatigueCols]:
    """player_matches plus pre-match fatigue columns (see FATIGUE_COLS)."""
    minutes = pl.col(FatigueInputCols.minutes)
    played = minutes.is_not_null().cast(pl.Int64)
    pm = (
        player_matches.join(
            _team_game_index(player_matches),
            on=[FatigueInputCols.team_id, FatigueInputCols.game_id],
            validate="m:1",
        )
        .sort(FatigueInputCols.player_id, FatigueInputCols.match_date)
        .with_columns(
            pl.col(FatigueInputCols.match_date)
            .diff()
            .dt.total_days()
            .over(FatigueInputCols.player_id)
            .alias(PlayerFatigueCols.rest_days),
            _prev_window(minutes, "7d").alias(PlayerFatigueCols.min_7d),
            _prev_window(played, "7d").alias(PlayerFatigueCols.n_7d),
            _prev_window(minutes, "28d").alias(PlayerFatigueCols.min_28d),
            # A new streak starts when the player missed the team's previous game,
            # or at his first game for a team (a transfer starts a new run).
            (pl.col(TEAM_GAME_NO).diff().over(_STREAK_KEYS) != 1)
            .fill_null(True)
            .alias(NEW_STREAK),
        )
        .with_columns(
            pl.col(NEW_STREAK).cast(pl.Int32).cum_sum().over(_STREAK_KEYS).alias(STREAK_ID)
        )
        .with_columns(
            pl.int_range(pl.len())
            .over(*_STREAK_KEYS, STREAK_ID)
            .alias(PlayerFatigueCols.streak_prev)
        )
    )
    out = pm.with_columns(_fatigue_score().round(1).alias(PlayerFatigueCols.fatigue_score)).drop(
        TEAM_GAME_NO, NEW_STREAK, STREAK_ID
    )
    return PlayerFatigueCols.validate(out, lazy=True)

    def player_matches(
    player_stat: DataFrame[PlayerStatCols], matches: DataFrame[CleanedMatchCols]
) -> DataFrame[FatigueInputCols]:
    """player_stat with each match's date and the player's side, one row per player per match.

    player_stat has no date or side, so both come from the match: match_date is the
    date part of the kick-off, and a player is home when his team is the home_team.
    Duplicate (game_id, player_id) rows, which Understat repeats for a few games, are dropped.
    """
    match_info: DataFrame[MatchSideCols] = MatchSideCols.validate(
        matches.select(
            pl.col(CleanedMatchCols.game_id).alias(MatchSideCols.game_id),
            pl.col(CleanedMatchCols.date).dt.date().alias(MatchSideCols.match_date),
            pl.col(CleanedMatchCols.home_team).alias(MatchSideCols.home_team),
        )
    )
    return FatigueInputCols.validate(
        player_stat.unique(
            [PlayerStatCols.game_id, PlayerStatCols.player_id], keep="first", maintain_order=True
        )
        .join(match_info, on=PlayerStatCols.game_id, how="inner", validate="m:1")
        .with_columns(
            (pl.col(PlayerStatCols.team) == pl.col(MatchSideCols.home_team)).alias(
                FatigueInputCols.is_home
            )
        )
        .drop(MatchSideCols.home_team)
    )

    def team_fatigue(fatigue: DataFrame[PlayerFatigueCols]) -> DataFrame[TeamFatigueCols]:
    """Mean starter fatigue per side, one row per game_id.

    A side with no starters recorded gets null in all of its columns, not 0.
    """
    starters = fatigue.filter(pl.col(PlayerFatigueCols.position) != "Sub")
    home, away = pl.col(PlayerFatigueCols.is_home), ~pl.col(PlayerFatigueCols.is_home)
    score, min_7d = pl.col(PlayerFatigueCols.fatigue_score), pl.col(PlayerFatigueCols.min_7d)
    out = (
        starters.group_by(PlayerFatigueCols.game_id)
        .agg(
            score.filter(home).mean().alias(TeamFatigueCols.h_fatigue),
            score.filter(away).mean().alias(TeamFatigueCols.a_fatigue),
            pl.when(home.any()).then(min_7d.filter(home).sum()).alias(TeamFatigueCols.h_min_7d),
            pl.when(away.any()).then(min_7d.filter(away).sum()).alias(TeamFatigueCols.a_min_7d),
        )
        .with_columns(
            (pl.col(TeamFatigueCols.h_fatigue) - pl.col(TeamFatigueCols.a_fatigue)).alias(
                TeamFatigueCols.fatigue_diff
            )
        )
    )
    return TeamFatigueCols.validate(out, lazy=True)

    money_printer_research.features.calendar ¶

    Plan A: every club's full match calendar (league, cups, Europe) from Transfermarkt.

    Understat only has league games, so a team that played in Europe on Wednesday looks rested for Saturday. Transfermarkt games (match_leagues) have all competitions. club_calendar computes pre-match schedule features on that full calendar, and team_schedule_features attaches them to each Understat game as h_ / a_ columns, through the tm_game_id that join_match_w_stat put on cleaned_matches.

    All features use only games before the match, except the look-ahead ones (days_to_next_all, next_is_europe), which use fixtures already scheduled.

    MAX_GAP_DAYS = 30 module-attribute ¶

    Cap on rest_days_all and days_to_next_all: beyond a month every club is fully rested, and the raw gaps reach years for clubs that spent seasons outside the five leagues. A club's first game in the data, with no previous one, gets MAX_GAP_DAYS of rest.

    def club_calendar(tm: DataFrame[MatchLeaguesCols]) -> DataFrame[ClubCalendarCols]:
    """One row per club per game with the schedule features, computed on all competitions."""
    sides = [(MatchLeaguesCols.home_club_id, True), (MatchLeaguesCols.away_club_id, False)]
    cal = pl.concat(
        [
            tm.select(
                pl.col(MatchLeaguesCols.tm_game_id).alias(ClubCalendarCols.tm_game_id),
                pl.col(MatchLeaguesCols.date).alias(ClubCalendarCols.date),
                pl.col(MatchLeaguesCols.comp_type).alias(ClubCalendarCols.comp_type),
                pl.col(own).alias(ClubCalendarCols.club_id),
                pl.lit(is_home).alias(ClubCalendarCols.is_home),
            )
            for own, is_home in sides
        ]
    ).sort(ClubCalendarCols.club_id, ClubCalendarCols.date, ClubCalendarCols.tm_game_id)

    comp, date = pl.col(ClubCalendarCols.comp_type), pl.col(ClubCalendarCols.date)
    any_game = date.is_not_null()
    europe_away = (comp == "europe") & ~pl.col(ClubCalendarCols.is_home)
    out = cal.with_columns(
        date.diff()
        .dt.total_days()
        .over(ClubCalendarCols.club_id)
        .clip(0, MAX_GAP_DAYS)
        .fill_null(MAX_GAP_DAYS)  # first game in the data: no previous one, fully rested
        .alias(ClubCalendarCols.rest_days_all),
        _prior_count(7, any_game).alias(ClubCalendarCols.games_7d_all),
        _prior_count(14, any_game).alias(ClubCalendarCols.games_14d_all),
        _prior_count(14, comp != "league").alias(ClubCalendarCols.non_league_14d),
        europe_away.shift(1)
        .over(ClubCalendarCols.club_id)
        .fill_null(False)
        .alias(ClubCalendarCols.prev_europe_away),
        (date.shift(-1).over(ClubCalendarCols.club_id) - date)
        .dt.total_days()
        .clip(0, MAX_GAP_DAYS)
        .alias(ClubCalendarCols.days_to_next_all),
        (comp.shift(-1).over(ClubCalendarCols.club_id) == "europe")
        .fill_null(False)
        .alias(ClubCalendarCols.next_is_europe),
    )
    return ClubCalendarCols.validate(out, lazy=True)

    def player_form(fatigue: DataFrame[PlayerFatigueCols]) -> pl.DataFrame:
    """Add prev_xg, prev_xa, prev_min and the per-90 rates xg90, xa90 and form90.

    def player_slots(form: pl.DataFrame) -> pl.DataFrame:
    """One row per game_id: each side's starters, ordered GK first, as slot columns."""
    starters = (
        form.filter(pl.col(FatigueInputCols.position) != "Sub")
        .with_columns(
            pl.when(pl.col(FatigueInputCols.is_home))
            .then(pl.lit("h"))
            .otherwise(pl.lit("a"))
            .alias("side"),
            pl.col(FatigueInputCols.position)
            .replace_strict({p: i for i, p in enumerate(POSITION_ORDER)})
            .alias("_order"),
        )
        .sort(FatigueInputCols.game_id, "side", "_order", FatigueInputCols.player_id)
        .with_columns(
            (pl.int_range(pl.len()).over(FatigueInputCols.game_id, "side") + 1).alias("slot")
        )
        .filter(pl.col("slot") <= N_SLOTS)
    )
    long = starters.unpivot(
        index=[FatigueInputCols.game_id, "side", "slot"],
        on=list(PLAYER_SLOT_FEATURES),
        variable_name="feature",
    ).with_columns(
        pl.format("{}_player{}_{}", "side", "slot", "feature").alias("column"),
        pl.col("value").cast(pl.Float64),
    )
    wide = long.pivot(on="column", index=FatigueInputCols.game_id, values="value")
    expected = list(slot_columns())
    missing = [pl.lit(None, pl.Float64).alias(c) for c in expected if c not in wide.columns]
    return wide.with_columns(missing).select(FatigueInputCols.game_id, *expected)

    money_printer_research.features.missing ¶

    Missing values in match_features, as their own table: missing.csv, one row per null cell.

    match_features keeps its nulls and gains no columns. missing_table lists every null cell (game_id, column) with the side it belongs to and the reason it is null:

    no_lineup         fatigue and the player slots, for a side Understat has no starters for
short_lineup      a player slot left empty because fewer than 11 starters are recorded
too_few_minutes   a starter's per-90 rates, below MIN_PREV_MIN previous minutes
no_schedule       the schedule features, for a side whose club has no Transfermarkt game
no_next_game      days_to_next_all, for a club's last game in the Transfermarkt data
no_tm_game        the Transfermarkt columns, for a match with no Transfermarkt game
no_formation      a side's formation columns, when its Transfermarkt game has no line-up
plain_shape       home_variant / away_variant, for a shape with no variant ("4-2-3-1")
other             none of these, such as a formation that did not parse

    (A club's first game has no previous one either; club_calendar gives it MAX_GAP_DAYS of rest rather than null, since the cap already means "a month or more".)

    The nulls are kept: filling them with a value computed from the table (a mean, a median) would leak later seasons into earlier ones, and a constant would pass for a real reading. reason_counts summarises missing.csv, and null_report counts nulls per feature group.

    @returns(MissingCols)
def missing_table(match_features: pl.DataFrame) -> pl.DataFrame:
    """One row per null cell of match_features: game_id, column, side and reason (MissingCols)."""
    key = CleanedMatchCols.game_id
    cells = (
        match_features.select(
            key, *(pl.col(c).is_null().alias(c) for c in match_features.columns if c != key)
        )
        .unpivot(index=key, variable_name="column", value_name="is_null")
        .filter(pl.col("is_null"))
        .drop("is_null")
    )
    sources = match_features.select(
        key,
        *(pl.col(f"{s}_{LINEUP_PROBE}").is_null().alias(f"{s}_no_lineup") for s in SIDES),
        *(pl.col(f"{s}_{SCHEDULE_PROBE}").is_null().alias(f"{s}_no_schedule") for s in SIDES),
        pl.col(CleanedMatchCols.tm_game_id).is_null().alias("no_tm_game"),
        *(
            pl.col(f"{p}_formation").is_null().alias(f"{s}_no_formation")
            for s, p in (("h", "home"), ("a", "away"))
        ),
    )
    column = pl.col("column")
    side = pl.col("side")
    per_side = {
        name: pl.when(side == "h").then(pl.col(f"h_{name}")).otherwise(pl.col(f"a_{name}"))
        for name in ("no_lineup", "no_schedule", "no_formation")
    }
    is_slot = pl.col("slot").is_not_null()
    empty_slot = (pl.col("feature") == EMPTY_SLOT_PROBE).any().over(key, "side", "slot") & is_slot
    either_no_lineup = pl.col("h_no_lineup") | pl.col("a_no_lineup")
    reason = (
        pl.when(is_slot & per_side["no_lineup"])
        .then(pl.lit("no_lineup"))
        .when(empty_slot)
        .then(pl.lit("short_lineup"))
        .when(is_slot & pl.col("feature").is_in(PER90))
        .then(pl.lit("too_few_minutes"))
        .when(column.is_in(FATIGUE_COLUMNS) & side.is_not_null() & per_side["no_lineup"])
        .then(pl.lit("no_lineup"))
        .when((column == "fatigue_diff") & either_no_lineup)
        .then(pl.lit("no_lineup"))
        .when(column.is_in(SCHEDULE_COLUMNS) & per_side["no_schedule"])
        .then(pl.lit("no_schedule"))
        .when(column.is_in(SCHEDULE_COLUMNS) & column.str.ends_with("_days_to_next_all"))
        .then(pl.lit("no_next_game"))
        .when(column.is_in(TM_COLUMNS) & pl.col("no_tm_game"))
        .then(pl.lit("no_tm_game"))
        .when(column.is_in(FORMATION_COLUMNS) & side.is_not_null() & per_side["no_formation"])
        .then(pl.lit("no_formation"))
        .when(column.str.ends_with("_variant") & column.is_in(FORMATION_COLUMNS))
        .then(pl.lit("plain_shape"))
        .otherwise(pl.lit("other"))
    )
    return (
        cells.join(sources, on=key, how="left")
        .with_columns(
            pl.coalesce(
                pl.when(pl.any_horizontal(column.str.starts_with(p) for p in prefixes)).then(
                    pl.lit(s)
                )
                for s, prefixes in _SIDE_PREFIX.items()
            ).alias("side"),
            column.str.extract(_SLOT, 1).alias("slot"),
            column.str.extract(_SLOT, 2).alias("feature"),
        )
        .with_columns(reason.alias("reason"))
        .select(key, "column", "side", "reason")
        .sort(key, maintain_order=True)
    )

    def null_report(df: pl.DataFrame) -> pl.DataFrame:
    """Nulls per feature group (see feature_group), most first; groups without nulls left out.

    share is the nulls over every cell of the group: rows times its number of columns.
    """
    counts = df.null_count().row(0, named=True)
    return (
        pl.DataFrame(
            {"group": [feature_group(c) for c in counts], "nulls": list(counts.values())},
            schema={"group": pl.String, "nulls": pl.Int64},
        )
        .group_by("group", maintain_order=True)
        .agg(pl.len().alias("columns"), pl.col("nulls").sum())
        .filter(pl.col("nulls") > 0)
        .with_columns(
            (pl.col("nulls") / (pl.col("columns") * max(df.height, 1))).round(4).alias("share")
        )
        .sort("nulls", descending=True, maintain_order=True)
    )
